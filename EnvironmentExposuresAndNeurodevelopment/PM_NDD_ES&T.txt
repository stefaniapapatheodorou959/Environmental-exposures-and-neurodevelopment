## 2) DEFINE EXPOSURE COLUMN MAPS AND MONTHLY INPUT

default_column_map <- function(outcome) {
  offset <- if (outcome == "ADHD") 1L else 0L
  list(pre_pm = (355:366) + offset, pre_temp = (367:378) + offset,
       pre_rh = (379:390) + offset, prenatal_pm = 235:271,
       prenatal_temp = 272:308, prenatal_rh = 309:345,
       post_pm = paste0("pm25.Month", 1:48),
       post_temp = paste0("tmpC.Month", 1:48), post_rh = 350:353)
}

month_anniversary <- function(date,nmonths) {
  yr <- as.integer(format(date,"%Y")); mo <- as.integer(format(date,"%m"))
  index <- yr*12L+mo-1L+nmonths
  first <- as.Date(sprintf("%04d-%02d-01",index%/%12L,index%%12L+1L))
  next_index <- index+1L
  next_first <- as.Date(sprintf("%04d-%02d-01",next_index%/%12L,next_index%%12L+1L))
  first+pmin(as.integer(format(date,"%d")),as.integer(next_first-first))-1L
}

join_monthly <- function(d,cfg) {
  if(is.null(cfg$monthly_long_file)) return(d)
  path <- cfg$monthly_long_file
  if(!grepl("^(/|[A-Za-z]:[/\\\\])",path)) path <- file.path(cfg$data_dir,path)
  long <- readRDS(path)
  need_columns(long,c(cfg$monthly_id,cfg$monthly_number,cfg$monthly_pm,cfg$monthly_temp))
  ids <- as.character(long[[cfg$monthly_id]])
  m <- as_num(long[[cfg$monthly_number]],cfg$monthly_number)
  if(anyNA(ids) || anyNA(m) || any(m!=floor(m) | m<1 | m>cfg$post_months))
    stop("Monthly file needs nonmissing IDs and integer months 1:48.")
  if(anyDuplicated(data.frame(ids,m))) stop("Duplicate child-month exposures; resolve before analysis.")
  position <- match(ids,as.character(d$patient_birth_ID)); keep <- !is.na(position)
  for(v in c("pm","temp")) {
    source <- if(v=="pm") cfg$monthly_pm else cfg$monthly_temp
    prefix <- if(v=="pm") "pm25.Month" else "tmpC.Month"
    cols <- paste0(prefix,seq_len(cfg$post_months))
    if(any(cols %in% names(d))) stop("Both wide and long monthly inputs exist. Choose one source.")
    x <- as_num(long[[source]],source)
    wide <- matrix(NA_real_,nrow(d),cfg$post_months)
    wide[cbind(position[keep],m[keep])] <- x[keep]
    colnames(wide) <- cols; d <- cbind(d,as.data.frame(wide))
  }
  d
}

## 3) DEFINE FOLLOW-UP AND RACE

need_columns <- function(d, vars) {
  missing <- setdiff(vars, names(d))
  if (length(missing)) stop("Missing columns: ", paste(missing, collapse = ", "))
}
as_num <- function(x, label) {
  if (is.factor(x)) x <- as.character(x)
  y <- suppressWarnings(as.numeric(x))
  if (any(!is.na(x) & !is.finite(y))) stop("Invalid numeric values in ", label)
  y
}
as_date <- function(x, label) {
  if (inherits(x, "Date")) return(x)
  if (inherits(x, "POSIXt")) return(as.Date(x))
  if (is.numeric(x)) stop("Convert ", label, " to Date with the verified origin first.")
  y <- as.Date(as.character(x))
  if (any(!is.na(x) & is.na(y))) stop("Invalid date in ", label)
  y
}
pick_column <- function(d, candidates, label) {
  hits <- intersect(candidates, names(d))
  if (!length(hits)) stop("Missing ", label, ": expected ", paste(candidates, collapse=" or "))
  hits[1L]
}
event_numbers <- c(ASD=1L, ADHD=3L, LDif=5L, DSLD=6L, DCD=7L, IntD=9L, BDis=12L)

prepare_followup <- function(d, outcome, cfg) {
  if (!outcome %in% names(event_numbers)) stop("Unknown outcome: ", outcome)
  if (!is.data.frame(d) || !nrow(d) || anyDuplicated(names(d))) stop("Invalid input data frame.")
  need_columns(d, c("patient_birth_ID", "birth_yr", "FIPS", "ZIP_Index", "Infant_Gender_Male"))
  if (anyNA(d$patient_birth_ID) || anyDuplicated(d$patient_birth_ID)) stop("Expected one row per child.")
  number <- event_numbers[[outcome]]
  flag <- paste0("Event_", number, "_MAIN")
  duration <- paste0("FU_Event_", number, "_MAIN")
  need_columns(d, c(flag, duration))
  birth_name <- "Deliv_Date"
  need_columns(d, birth_name)
  enrollment_name <- pick_column(d, c("Infant_length_enrollment.x", "Infant_length_enrollment"), "enrollment duration")
  d$Deliv_Date <- as_date(d[[birth_name]], birth_name)
  enrollment <- as_num(d[[enrollment_name]], enrollment_name)
  status <- as_num(d[[flag]], flag)
  diagnosis <- as_num(d[[duration]], duration)
  if (anyNA(d$Deliv_Date) || anyNA(enrollment) || anyNA(status) ||
      any(!status %in% 0:1) || any(enrollment < 0)) stop("Invalid birth date, enrollment, or event flag.")
  if (any(status == 1 & (is.na(diagnosis) | diagnosis <= 0))) stop("Cases need positive diagnosis times: ", duration)
  admin <- as.numeric(cfg$study_end - d$Deliv_Date)
  death <- rep(Inf, nrow(d))
  if (!is.null(cfg$death_date_column)) {
    need_columns(d, cfg$death_date_column)
    dates <- as_date(d[[cfg$death_date_column]], cfg$death_date_column)
    death[!is.na(dates)] <- as.numeric(dates[!is.na(dates)] - d$Deliv_Date[!is.na(dates)])
    if (any(death < 0)) stop("Death precedes birth.")
  }
  end <- pmin(enrollment, admin, death)
  d$.event <- as.integer(status == 1 & !is.na(diagnosis) & diagnosis <= end)
  d$.stop <- ifelse(d$.event == 1, diagnosis, end)
  d$.start <- 0
  d$.censor <- as.integer(d$.event == 0 & pmin(enrollment, death) < admin)
  d$.censor_definition <- "early_enrollment_or_death"
  reason_col <- if(outcome %in% names(cfg$exit_reason_columns))
    cfg$exit_reason_columns[[outcome]] else NULL
  if (!is.null(reason_col)) {
    need_columns(d, reason_col)
    if (any(d$.censor == 1 & is.na(d[[reason_col]]))) stop("Early exits have missing exit reasons.")
    d$.censor <- as.integer(d$.censor == 1 & d[[reason_col]] %in% cfg$censor_reason_values)
    d$.censor_definition <- "specified_exit_reasons"
  }
  d[[paste0("Event_", number, "_FU_correct")]] <- d$.stop
  d[[paste0("Event_", number, "_MAIN_correct")]] <- d$.event
  d[[paste0(outcome, "_time")]] <- d$.stop
  if (!"race" %in% names(d)) {
    indicators <- c("Race_WHITE", "Race_BLACK_OR_AFRICAN_AMERICAN", "Race_HISPANIC_OR_LATINO",
                    "Race_ASIAN_OR_OTHER_PACIFIC", "Race_UNKNOWN_OTHER")
    need_columns(d, indicators)
    r <- sapply(indicators, function(v) as_num(d[[v]], v))
    if (any(!is.na(r) & !r %in% 0:1)) stop("Race indicators must be 0/1.")
    good <- complete.cases(r)
    if (any(rowSums(r[good,,drop=FALSE]) != 1)) stop("Resolve overlapping/unclassified race indicators.")
    d$race <- NA_character_
    d$race[good] <- sub("^Race_", "", indicators[max.col(r[good,,drop=FALSE], ties.method="first")])
  }
  d$race <- factor(d$race)
  d$Infant_Gender_Male <- as_num(d$Infant_Gender_Male, "Infant_Gender_Male")
  if (any(!is.na(d$Infant_Gender_Male) & !d$Infant_Gender_Male %in% 0:1)) stop("Sex flag must be 0/1.")
  if ("season_conception" %in% names(d)) d$season_conception <- factor(d$season_conception)
  if ("UA" %in% names(d)) d$UA <- factor(d$UA)
  if ("region" %in% names(d)) d$region <- factor(d$region)
  numeric_covariates <- intersect(c(setdiff(base_covariates,"season_conception"),"ndvi",
                                    "Infant_Gender_Unknown"),names(d))
  for(v in numeric_covariates) d[[v]] <- as_num(d[[v]],v)
  binary_covariates <- intersect(c("Alcohol_abuse","Smoking","Substance_abuse",
    "Poor_nutrition","prenatal_folate","Infant_Gender_Unknown"),names(d))
  for(v in binary_covariates) if(any(!is.na(d[[v]]) & !d[[v]] %in% 0:1)) stop("Expected 0/1/NA in ",v)
  d$.id <- seq_len(nrow(d))
  d$.zip <- d$ZIP_Index
  d$.stratum <- interaction(d$birth_yr, d$FIPS, d$race, d$Infant_Gender_Male,
                            drop=TRUE, lex.order=TRUE)
  attr(d, "followup_audit") <- data.frame(outcome=outcome, input_n=nrow(d),
    raw_events=sum(status), observed_events=sum(d$.event),
    raw_cases_outside_observation=sum(status == 1 & d$.event == 0),
    no_positive_followup=sum(d$.stop <= 0),
    noncase_enrollment_past_end=sum(status == 0 & enrollment > admin),
    study_end=as.character(cfg$study_end), enrollment_column=enrollment_name)
  d
}

## 4) BUILD THE EXPOSURE MATRICES ONCE PER OUTCOME

exposure_matrices <- function(d, outcome, cfg) {
  mapping <- cfg$column_maps[[outcome]]
  if (is.null(mapping)) mapping <- default_column_map(outcome)
  if(!is.null(cfg$monthly_long_file)) {
    mapping$post_pm <- paste0("pm25.Month",1:48)
    mapping$post_temp <- paste0("tmpC.Month",1:48)
  }
  mats <- list(); audit <- list()
  for (key in names(mapping)) {
    cols <- mapping[[key]]
    expected <- if (startsWith(key,"pre_")) 12L else if(startsWith(key,"prenatal_")) 37L else if(key %in% c("post_pm","post_temp")) 48L else 4L
    if (length(cols) != expected) stop(key, " requires ", expected, " ordered columns.")
    if (is.numeric(cols)) {
      if (any(cols < 1 | cols > ncol(d) | cols != as.integer(cols))) stop("Invalid positions for ",key)
      cols <- names(d)[cols]
    }
    need_columns(d, cols)
    if (anyDuplicated(cols)) stop("Repeated column in ", key)
    pattern <- if (grepl("_pm$",key)) "pm" else if(grepl("_temp$",key)) "tmp|temp" else "rh|humid"
    if (any(!grepl(pattern, cols, ignore.case=TRUE))) stop("Unexpected columns in ",key,": ",paste(cols,collapse=", "),". Set CFG$column_maps explicitly.")
    if (!all(vapply(d[cols], is.numeric, logical(1)))) stop("Nonnumeric exposure columns in ", key)
    mat <- as.matrix(d[cols])
    if (any(!is.finite(mat) & !is.na(mat))) stop("Infinite exposures in ",key)
    if (grepl("_pm$",key) && any(mat < 0, na.rm=TRUE)) stop("Negative PM exposure in ",key)
    if (grepl("_rh$",key) && any(mat < 0 | mat > 100, na.rm=TRUE)) stop("RH outside 0-100 in ",key)
    mats[[key]] <- mat
    audit[[key]] <- data.frame(outcome=outcome, block=key, position=seq_along(cols), column=cols)
  }
  need <- c("pre_pm","pre_temp","pre_rh","prenatal_pm","prenatal_temp","prenatal_rh","post_pm","post_temp","post_rh")
  if (!all(need %in% names(mats))) stop("Incomplete exposure mapping.")
  names_all <- unlist(lapply(mats,colnames),use.names=FALSE)
  if(anyDuplicated(names_all)) stop("Exposure blocks overlap. Correct the column map.")
  for(key in c("post_pm","post_temp")) {
    x <- mats[[key]]; ok <- which(complete.cases(x))
    if(length(ok)) {
      z <- x[ok[seq_len(min(1000,length(ok)))],,drop=FALSE]
      copied <- vapply(seq(1,48,12),function(k)
        all(z[,k:(k+11),drop=FALSE]==z[,k]),logical(1))
      if(all(copied)) stop(key," appears to contain annual values repeated across months. Supply actual monthly data.")
    }
  }
  list(matrices=mats, audit=do.call(rbind,audit))
}

## 5) SET COVARIATES AND COMPLETE-CASE SAMPLES

base_covariates <- c("season_conception", "Age", "Alcohol_abuse", "Smoking", "Substance_abuse",
  "Poor_nutrition", "prenatal_folate", "medhouseholdincome", "pct_owner_occ", "medianhousevalue",
  "education", "poverty", "pop_density_30sec_NASA", "smoke_rate", "mean_bmi",
  "nearest_hospital_km", "pct_amb_visit")

make_analysis <- function(d, exposure, outcome, period, cfg, subgroup=FALSE, age3=FALSE) {
  e <- exposure$matrices
  covariates <- base_covariates
  pre <- period == "preconception_prenatal"
  if(pre && outcome %in% cfg$ndvi_pre_outcomes) covariates <- c(covariates,"ndvi")
  if(!pre && "Infant_Gender_Unknown" %in% names(d)) covariates <- c(covariates,"Infant_Gender_Unknown")
  pm_pre <- cbind(e$pre_pm,e$prenatal_pm)
  if(pre) {
    X <- list(pm=pm_pre, temp=cbind(e$pre_temp,e$prenatal_temp), rh=cbind(e$pre_rh,e$prenatal_rh))
    lag <- list(pm=c(-48,0),temp=c(-48,0),rh=c(-48,0))
    df <- if(subgroup) cfg$subgroup_pre_lag_df else cfg$pre_lag_df
    start <- rep(0,nrow(d))
  } else {
    X <- list(pm=e$post_pm,temp=e$post_temp,rh=e$post_rh)
    lag <- list(pm=c(-47,0),temp=c(-47,0),rh=c(-3,0))
    if(cfg$post_resolution != "monthly" || ncol(X$pm)!=48 || ncol(X$temp)!=48 || ncol(X$rh)!=4)
      stop("Postnatal model requires 48 actual PM/temp months and four annual RH values.")
    df <- cfg$post_lag_df
    if(cfg$post_adjustment == "averages") {
      d$preconceptionPMmean <- rowMeans(e$pre_pm)
      d$prenatalPMmean <- rowMeans(e$prenatal_pm)
      covariates <- c(covariates,"preconceptionPMmean","prenatalPMmean")
    } else if(cfg$post_adjustment == "precon_prenatal_basis") X$pre_pm <- pm_pre
    else stop("Unknown post_adjustment.")
    if(cfg$post_followup == "after_exposure_window") {
      if(cfg$post_year_definition != "age_year") stop("Monthly inputs must be age-aligned.")
      entry_date <- month_anniversary(d$Deliv_Date,48L)
      start <- as.numeric(entry_date-d$Deliv_Date)
    } else if(cfg$post_followup == "legacy_birth_origin") {
      start <- rep(0,nrow(d))
    } else stop("Unknown post_followup.")
  }
  d$.start <- start
  need <- unique(c(covariates,".id",".zip",".stratum",".start",".stop",".event",".censor"))
  need_columns(d,need)
  keep <- complete.cases(d[need]) & d$.stop > d$.start
  for(x in X) keep <- keep & complete.cases(x)
  if(age3) keep <- keep & !(d$.event == 1 & d$.stop < 1095)
  dat <- droplevels(d[keep,need,drop=FALSE])
  X <- lapply(X,function(x) x[keep,,drop=FALSE])
  if(!nrow(dat) || !sum(dat$.event)) stop("No eligible outcome events for ",outcome," / ",period)
  numerical <- vapply(dat,is.numeric,logical(1))
  if(any(!is.finite(as.matrix(dat[numerical])))) stop("Nonfinite covariate values.")
  constant <- covariates[vapply(dat[covariates],function(x) length(unique(x)) < 2L,logical(1))]
  covariates <- setdiff(covariates,constant)
  list(data=dat,X=X,lag=lag,df=df,covariates=covariates,constant=constant,
       input_n=nrow(d),excluded_n=sum(!keep),outcome=outcome,period=period,
       exposure_df=if(pre) cfg$pre_exposure_df else NULL)
}

## 6) DEFINE CROSS-BASES AND FIT COX MODELS

make_bases <- function(a,cfg) {
  basis <- list()
  for(v in c("pm","temp","rh")) {
    edf <- if(is.null(a$exposure_df)) NULL else if(length(a$exposure_df)==1L)
      a$exposure_df else a$exposure_df[[v]]
    exposure_arg <- if(is.null(edf)) list(fun="lin") else list(fun="ns",df=edf)
    if(a$df[[v]] > ncol(a$X[[v]])) stop("Lag df exceeds distinct observed exposure periods.")
    basis[[v]] <- dlnm::crossbasis(a$X[[v]],lag=a$lag[[v]],argvar=exposure_arg,
                                 arglag=list(fun="ns",df=a$df[[v]]))
  }
  if("pre_pm" %in% names(a$X)) {
    p <- cfg$post_pre_exposure_df[[a$outcome]]
    basis$pre_pm <- dlnm::crossbasis(a$X$pre_pm,lag=c(-48,0),
      argvar=if(is.null(p)) list(fun="lin") else list(fun="ns",df=p),
      arglag=list(fun="ns",df=6))
  }
  basis
}
expand_basis <- function(cb,index) {
  attr_old <- attributes(cb); ans <- cb[index,,drop=FALSE]
  for(n in setdiff(names(attr_old),c("dim","dimnames"))) attr(ans,n) <- attr_old[[n]]
  ans
}

fit_analysis <- function(a,cfg,intervals=NULL,predictions=TRUE) {
  basis <- make_bases(a,cfg)
  dat <- a$data
  if(is.null(intervals)) dat$.weight <- 1 else {
    ii <- match(intervals$.id,dat$.id)
    if(anyNA(ii)) stop("IPCW interval IDs do not match analysis rows.")
    dat <- dat[ii,,drop=FALSE]
    dat$.start <- intervals$.start; dat$.stop <- intervals$.stop
    dat$.event <- intervals$.event; dat$.weight <- intervals$.weight
    basis <- lapply(basis,expand_basis,index=ii)
  }
  cb_pm <- basis$pm; cb_temp <- basis$temp; cb_rh <- basis$rh
  terms <- c("cb_pm","cb_temp","cb_rh",a$covariates,"strata(.stratum)")
  if(!is.null(basis$pre_pm)) { cb_pre_pm <- basis$pre_pm; terms <- c(terms,"cb_pre_pm") }
  formula <- as.formula(paste("Surv(.start,.stop,.event) ~",paste(terms,collapse=" + ")))
  model <- survival::coxph(formula,data=dat,weights=.weight,cluster=.zip,robust=TRUE,
    ties="efron",na.action=na.fail,singular.ok=FALSE,x=FALSE,y=FALSE,model=FALSE)
  if(any(!is.finite(coef(model)))) stop("Nonfinite coefficients; inspect convergence/strata.")
  if(!predictions) return(list(AIC=AIC(model),n=nrow(a$data),events=sum(a$data$.event)))
  pm <- dlnm::crosspred(cb_pm,model,cen=cfg$reference[["pm"]],at=cfg$comparison[["pm"]])
  temp <- dlnm::crosspred(cb_temp,model,cen=cfg$reference[["temp"]],at=cfg$comparison[["temp"]])
  pm_curve <- dlnm::crosspred(cb_pm,model,cen=cfg$reference[["pm"]],at=0:30)
  temp_curve <- dlnm::crosspred(cb_temp,model,cen=cfg$reference[["temp"]],at=0:40)
  windows <- if(a$period=="postnatal") list(months_1_12=c(-47,-36),
    months_13_24=c(-35,-24),months_25_36=c(-23,-12),months_37_48=c(-11,0)) else
    list(preconception_12_weeks=c(-48,-37),prenatal_37_weeks=c(-36,0))
  window_rows <- list()
  for(v in c("pm","temp")) for(w in names(windows)) {
    p <- if(v=="pm") dlnm::crosspred(cb_pm,model,cen=cfg$reference[[v]],
      at=cfg$comparison[[v]],lag=windows[[w]]) else
      dlnm::crosspred(cb_temp,model,cen=cfg$reference[[v]],at=cfg$comparison[[v]],lag=windows[[w]])
    window_rows[[paste(v,w)]] <- data.frame(exposure=v,window=w,
      reference=cfg$reference[[v]],comparison=cfg$comparison[[v]],
      HR=unname(p$allRRfit),LCL=unname(p$allRRlow),UCL=unname(p$allRRhigh))
  }
  ans <- list(pm=pm,temp=temp,pm_curve=pm_curve,temp_curve=temp_curve,
    window_HRs=do.call(rbind,window_rows),
    n=nrow(a$data),events=sum(a$data$.event),intervals=nrow(dat),AIC=AIC(model),
    dropped_constant=a$constant,basis=lapply(basis,function(b) attributes(b)[c("argvar","arglag","lag")]))
  ans
}

## 7) EXTRACT HAZARD RATIOS AND DRAW DLNM PLOTS

result_table <- function(fit,a,cfg,label) {
  do.call(rbind,lapply(c("pm","temp"),function(v) {
    p <- fit[[v]]
    data.frame(outcome=a$outcome,period=a$period,analysis=label,N=fit$n,events=fit$events,
      exposure=v,reference=cfg$reference[[v]],comparison=cfg$comparison[[v]],
      HR=unname(p$allRRfit[1]),LCL=unname(p$allRRlow[1]),UCL=unname(p$allRRhigh[1]),
      exposure_function=if(is.null(a$exposure_df)) "linear" else "natural_spline",
      exposure_df=if(is.null(a$exposure_df)) 1 else if(length(a$exposure_df)==1L) a$exposure_df else a$exposure_df[[v]],
      lag_df=a$df[[v]],
      AIC=fit$AIC,post_resolution=if(a$period=="postnatal") cfg$post_resolution else NA_character_,
      post_followup=if(a$period=="postnatal") cfg$post_followup else NA_character_)
  }))
}
save_plots <- function(fit,a,cfg,prefix) {
  pdf(paste0(prefix,"_plots.pdf"),width=10,height=4.5)
  on.exit(dev.off(),add=TRUE)
  for(v in c("pm","temp")) {
    par(mfrow=c(1,2),mar=c(4,4,3,1))
    plot(fit[[paste0(v,"_curve")]],"overall",ci="area",ylab="Cumulative HR",
      xlab=if(v=="pm") "PM2.5 (ug/m3)" else "Temperature (C)",main=paste(a$outcome,v))
    abline(h=1,lty=2)
    plot(fit[[v]],"slices",var=cfg$comparison[[v]],ci="area",xaxt="n",
      xlab=if(a$period=="preconception_prenatal") "Week relative to conception" else "Postnatal month",
      ylab=paste0("HR: ",cfg$comparison[[v]]," versus ",cfg$reference[[v]]), main=a$outcome)
    if(a$period=="preconception_prenatal") axis(1,at=c(-48,-36,-26,-16,-6,0),labels=c(-12,0,10,20,30,36))
    else if(diff(a$lag[[v]])==47) axis(1,at=c(-47,-36,-24,-12,0),labels=c(1,12,24,36,48))
    else axis(1,at=-3:0,labels=0:3)
    abline(h=1,lty=2)
  }
}

## 8) DEFINE AIC COMPARISONS

compare_aic <- function(a,cfg) {
  grid <- expand.grid(pm=cfg$aic_pm_lag_df,temp=cfg$aic_other_lag_df,rh=cfg$aic_other_lag_df)
  rows <- lapply(seq_len(nrow(grid)),function(i) {
    b <- a; b$df <- unlist(grid[i,])
    warnings <- character()
    ans <- tryCatch(withCallingHandlers(fit_analysis(b,cfg,predictions=FALSE),
      warning=function(w) {warnings <<- c(warnings,conditionMessage(w));invokeRestart("muffleWarning")}),
      error=function(e)e)
    data.frame(grid[i,],N=nrow(a$data),events=sum(a$data$.event),
      AIC=if(inherits(ans,"error")) NA_real_ else ans$AIC,
      error=if(inherits(ans,"error")) conditionMessage(ans) else "",
      warning=paste(unique(warnings),collapse=" | "))
  })
  out <- do.call(rbind,rows)
  if(!any(is.finite(out$AIC))) stop("All AIC candidates failed.")
  out$delta_AIC <- out$AIC-min(out$AIC,na.rm=TRUE)
  out
}

compare_exposure_aic <- function(a,cfg) {
  grid <- expand.grid(pm=cfg$aic_exposure_df,temp=cfg$aic_exposure_df,rh=cfg$aic_exposure_df)
  rows <- lapply(seq_len(nrow(grid)),function(i) {
    b <- a;b$exposure_df <- unlist(grid[i,]);b$df <- c(pm=4L,temp=4L,rh=4L)
    warnings <- character()
    ans <- tryCatch(withCallingHandlers(fit_analysis(b,cfg,predictions=FALSE),
      warning=function(w) {warnings <<- c(warnings,conditionMessage(w));invokeRestart("muffleWarning")}),
      error=function(e)e)
    data.frame(grid[i,],N=nrow(a$data),events=sum(a$data$.event),
      AIC=if(inherits(ans,"error")) NA_real_ else ans$AIC,
      error=if(inherits(ans,"error")) conditionMessage(ans) else "",
      warning=paste(unique(warnings),collapse=" | "))
  })
  out <- do.call(rbind,rows)
  if(!any(is.finite(out$AIC))) stop("All exposure-response AIC candidates failed.")
  out$delta_AIC <- out$AIC-min(out$AIC,na.rm=TRUE)
  out
}

## 9) DEFINE TIME-VARYING IPCW

estimate_ipcw <- function(a,cfg) {
  d <- a$data
  basis <- make_bases(a,cfg)
  basis_names <- character()
  for(v in names(basis)) {
    b <- as.data.frame(unclass(basis[[v]]))
    names(b) <- paste0(".cens_",v,seq_len(ncol(b)))
    basis_names <- c(basis_names,names(b))
    d <- cbind(d,b)
  }
  cov <- c(a$covariates,basis_names)
  fit_censor <- function(terms) {
    f <- as.formula(paste("Surv(.start,.stop,.censor) ~",paste(c("strata(.stratum)",terms),collapse=" + ")))
    withCallingHandlers(
      survival::coxph(f,data=d,ties="efron",model=TRUE,na.action=na.fail,singular.ok=FALSE),
      warning=function(w) stop("Censoring model warning: ",conditionMessage(w)))
  }
  numerator <- denominator <- NULL
  if(any(d$.censor==1L)) {
    numerator <- fit_censor(character())
    denominator <- fit_censor(cov)
  }
  subjects <- d[c(".id",".stratum",".start",".stop",".event")]
  subjects$.position <- seq_len(nrow(d))
  subjects$.entry <- subjects$.start
  pieces <- split(subjects,subjects$.stratum,drop=TRUE)
  total <- 0
  for(k in seq_along(pieces)) {
    z <- pieces[[k]]; cuts <- sort(unique(z$.stop[z$.event==1]))
    total <- total + sum(1L + findInterval(z$.stop,cuts)-findInterval(z$.start,cuts))
    if(total > cfg$ipcw_max_rows) stop("IPCW expansion exceeds CFG$ipcw_max_rows. Run one outcome at a time or raise the limit after checking RAM.")
    if(length(cuts)) z <- survival::survSplit(Surv(.start,.stop,.event)~.,
      data=z,cut=cuts,start=".start",end=".stop",event=".event")
    z$.used <- z$.stop %in% cuts
    pieces[[k]] <- z
  }
  it <- do.call(rbind,pieces)
  if(sum(it$.event)!=sum(d$.event) || any(it$.stop<=it$.start)) stop("Invalid IPCW splitting.")
  log_G <- function(fit) {
    if(is.null(fit)) return(rep(0,nrow(it)))
    h <- survival::basehaz(fit,centered=TRUE)
    labels <- as.character(fit$model[["strata(.stratum)"]])
    if(!"strata" %in% names(h)) h$strata <- unique(labels)
    curves <- split(h,as.character(h$strata),drop=TRUE)
    rows <- split(seq_len(nrow(it)),labels[it$.position])
    lp <- if(length(coef(fit))) fit$linear.predictors else rep(0,nrow(d))
    if(any(!is.finite(lp)) || !all(names(rows) %in% names(curves))) stop("Censoring strata do not align.")
    ans <- numeric(nrow(it))
    for(s in names(rows)) {
      j <- rows[[s]]; curve <- curves[[s]]
      before <- findInterval(it$.stop[j],curve$time,left.open=TRUE)
      entry <- findInterval(it$.entry[j],curve$time)
      H <- pmax(0,c(0,curve$hazard)[before+1L]-c(0,curve$hazard)[entry+1L])
      ans[j] <- -exp(log(H)+lp[it$.position[j]])
    }
    if(any(!is.finite(ans))) stop("Invalid predicted censoring survival.")
    ans
  }
  log_den <- log_G(denominator)
  it$.weight <- exp(log_G(numerator)-log_den)
  if(any(!is.finite(it$.weight)|it$.weight<=0)) stop("Invalid IPCW; inspect positivity and model fit.")
  used <- it$.used
  if(!any(used)) stop("No outcome risk sets for IPCW.")
  w <- it$.weight[used]
  cap <- as.numeric(quantile(w,cfg$ipcw_truncate_quantile,names=FALSE))
  trunc <- it
  trunc$.weight <- pmin(trunc$.weight,cap)
  riskset <- paste(it$.stratum[used],it$.stop[used],sep=":")
  diagnostics_one <- function(weights,type) {
    x <- weights[used]
    scaled <- x/ave(x,riskset,FUN=max)
    sums <- rowsum(cbind(n=1,w=scaled,w2=scaled^2),riskset,reorder=FALSE)
    ess <- sums[,"w"]^2/sums[,"w2"]
    q <- quantile(x,c(0,.5,.95,.99,1),names=FALSE)
    data.frame(type=type,N=nrow(d),events=sum(d$.event),censor_events=sum(d$.censor),
      intervals=nrow(it),mean=mean(x),CV=sd(x)/mean(x),min=q[1],median=q[2],p95=q[3],p99=q[4],max=q[5],
      min_G=exp(min(log_den[used])),min_riskset_ESS_fraction=min(ess/sums[,"n"]),
      median_riskset_ESS_fraction=median(ess/sums[,"n"]),
      truncation_quantile=cfg$ipcw_truncate_quantile,truncation_cap=cap,
      fraction_capped=mean(w>cap))
  }
  cols <- c(".id",".start",".stop",".event",".weight")
  if(any(log_den[used]<log(.01))) warning("Some censoring survival probabilities are below 0.01; inspect IPCW diagnostics.")
  list(raw=it[cols],truncated=trunc[cols],diagnostics=rbind(
    diagnostics_one(it$.weight,"untruncated"),diagnostics_one(trunc$.weight,"truncated")))
}

## 10) DEFINE COHORT SUMMARIES AND SUBGROUPS

save_descriptives <- function(d,outcome,outdir) {
  groups <- c("all","noncase","case")
  rows <- lapply(groups,function(g) {
    keep <- if(g=="all") rep(TRUE,nrow(d)) else if(g=="case") d$.event==1 else d$.event==0
    t <- d$.stop[keep & d$.stop>0]/365.25
    data.frame(outcome=outcome,group=g,N=sum(keep),positive_followup_n=length(t),
      mean_followup_years=if(length(t)) mean(t) else NA_real_,
      sd_followup_years=if(length(t)>1) sd(t) else NA_real_,
      median_followup_years=if(length(t)) median(t) else NA_real_)
  })
  write.csv(do.call(rbind,rows),file.path(outdir,paste0(outcome,"_followup_summary.csv")),row.names=FALSE)
  counts <- lapply(c("Infant_Gender_Male","race","UA","region"),function(v) {
    if(!v %in% names(d)) return(NULL)
    z <- as.data.frame(table(group=as.character(d[[v]]),event=d$.event,useNA="ifany"))
    z$variable <- v; z
  })
  write.csv(do.call(rbind,counts),file.path(outdir,paste0(outcome,"_cohort_counts.csv")),row.names=FALSE)
}

subset_inputs <- function(d,exposure,keep) {
  list(data=d[keep,,drop=FALSE],exposure=list(matrices=lapply(exposure$matrices,
    function(x) x[keep,,drop=FALSE]),audit=exposure$audit))
}

run_step <- function(label, code) {
  message(label)
  warnings <- character()
  ans <- tryCatch(withCallingHandlers(force(code), warning=function(w) {
    warnings <<- c(warnings, conditionMessage(w)); invokeRestart("muffleWarning")
  }), error=function(e) e)
  failed <- inherits(ans, "error")
  analysis_status[[length(analysis_status)+1L]] <<- data.frame(
    analysis=label, status=if(failed) "FAILED" else if(length(warnings)) "WARNING" else "OK",
    message=if(failed) conditionMessage(ans) else paste(unique(warnings), collapse=" | "))
  save_status()
  if (failed) message("FAILED: ", label, ": ", conditionMessage(ans))
  invisible(ans)
}

save_fit <- function(a, label, intervals=NULL) {
  fit <- fit_analysis(a, CFG, intervals)
  table <- result_table(fit, a, CFG, label)
  results[[length(results)+1L]] <<- table
  prefix <- file.path(CFG$output_dir, gsub("[^A-Za-z0-9_.-]", "_", label))
  write.csv(table, paste0(prefix, "_HR.csv"), row.names=FALSE)
  write.csv(fit$window_HRs, paste0(prefix, "_window_HRs.csv"), row.names=FALSE)
  write.csv(data.frame(input_n=a$input_n, analyzed_n=nrow(a$data), excluded_n=a$excluded_n,
    dropped_constant=paste(a$constant,collapse=";")), paste0(prefix,"_sample.csv"), row.names=FALSE)
  dput(fit$basis, file=paste0(prefix, "_basis.R"))
  saveRDS(fit, paste0(prefix, "_predictions.rds"))
  save_plots(fit, a, CFG, prefix)
  invisible(fit)
}

for (outcome in CFG$outcomes) {

  ## 12) READ ONE OUTCOME AND PREPARE EXPOSURES AND FOLLOW-UP

  prepared <- run_step(paste0(outcome, "_prepare"), {
    input_path <- CFG$input_files[[outcome]]
    if (!grepl("^(/|[A-Za-z]:[/\\\\])", input_path)) input_path <- file.path(CFG$data_dir,input_path)
    raw <- readRDS(input_path)
    raw <- join_monthly(raw, CFG)
    d <- prepare_followup(raw, outcome, CFG)
    ex <- exposure_matrices(d, outcome, CFG)
    write.csv(attr(d,"followup_audit"), file.path(CFG$output_dir,paste0(outcome,"_followup_audit.csv")), row.names=FALSE)
    write.csv(ex$audit, file.path(CFG$output_dir,paste0(outcome,"_exposure_mapping.csv")), row.names=FALSE)
    save_descriptives(d, outcome, CFG$output_dir)
    list(data=d, exposure=ex)
  })
  if (inherits(prepared,"error")) next
  d <- prepared$data
  ex <- prepared$exposure

  ## 13) FIT PRECONCEPTION, PRENATAL AND MONTHLY POSTNATAL MODELS

  for (period in c("preconception_prenatal", "postnatal")) {
    run_step(paste(outcome, period, "main", sep="_"), {
      a <- make_analysis(d, ex, outcome, period, CFG)
      save_fit(a, paste(outcome, period, "main", sep="_"))
    })
  }

  ## 14) FIT SEX, RACE, URBANICITY AND REGION SUBGROUPS

  if (CFG$run_subgroups) for (period in c("preconception_prenatal", "postnatal")) {
    for (group in CFG$subgroup_variables) {
      if (!group %in% names(d)) {
        run_step(paste(outcome,period,group,sep="_"), stop("Missing subgroup field: ",group))
        next
      }
      for (value in sort(unique(as.character(d[[group]][!is.na(d[[group]])])))) {
        label <- paste(outcome, period, group, value, sep="_")
        run_step(label, {
          keep <- which(!is.na(d[[group]]) & as.character(d[[group]])==value)
          sub <- subset_inputs(d, ex, keep)
          a <- make_analysis(sub$data, sub$exposure, outcome, period, CFG, subgroup=TRUE)
          save_fit(a, label)
        })
      }
    }
  }

  ## 15) RESTRICT DIAGNOSES TO AGE THREE OR OLDER

  if (CFG$run_age3 && outcome %in% CFG$age3_outcomes) {
    run_step(paste0(outcome,"_diagnoses_at_least_1095_days"), {
      a <- make_analysis(d, ex, outcome, "preconception_prenatal", CFG, age3=TRUE)
      save_fit(a, paste0(outcome,"_diagnoses_at_least_1095_days"))
    })
  }

  ## 16) FIT THE ASD NONLINEAR EXPOSURE-RESPONSE SENSITIVITY

  if (CFG$run_nonlinear && outcome %in% CFG$nonlinear_pre_outcomes) {
    run_step(paste0(outcome,"_nonlinear_2df"), {
      a <- make_analysis(d, ex, outcome, "preconception_prenatal", CFG)
      a$exposure_df <- 2L
      save_fit(a, paste0(outcome,"_nonlinear_2df"))
    })
  }

  ## 17) ADJUST POSTNATAL MODELS FOR THE EARLY-EXPOSURE CROSS-BASIS

  if (CFG$run_post_early_basis) {
    run_step(paste0(outcome,"_postnatal_early_exposure_basis"), {
      basis_cfg <- CFG
      basis_cfg$post_adjustment <- "precon_prenatal_basis"
      a <- make_analysis(d, ex, outcome, "postnatal", basis_cfg)
      save_fit(a, paste0(outcome,"_postnatal_early_exposure_basis"))
    })
  }

  ## 18) COMPARE SPLINE DEGREES OF FREEDOM USING AIC

  if (CFG$run_aic) for (period in c("preconception_prenatal", "postnatal")) {
    run_step(paste(outcome,period,"AIC",sep="_"), {
      a <- make_analysis(d, ex, outcome, period, CFG)
      write.csv(compare_aic(a,CFG), file.path(CFG$output_dir,paste0(outcome,"_",period,"_AIC.csv")), row.names=FALSE)
      if (period=="postnatal" && CFG$run_exposure_aic)
        write.csv(compare_exposure_aic(a,CFG), file.path(CFG$output_dir,paste0(outcome,"_postnatal_exposure_AIC.csv")), row.names=FALSE)
    })
  }

  ## 19) COMPARE UNWEIGHTED AND IPCW MODELS ON THE SAME SAMPLE

  if (CFG$run_ipcw) for (period in CFG$ipcw_periods) {
    run_step(paste(outcome,period,"IPCW",sep="_"), {
      a <- make_analysis(d, ex, outcome, period, CFG)
      weights <- estimate_ipcw(a, CFG)
      weights$diagnostics$censor_definition <- unique(d$.censor_definition)
      write.csv(weights$diagnostics, file.path(CFG$output_dir,paste0(outcome,"_",period,"_IPCW_diagnostics.csv")), row.names=FALSE)
      save_fit(a, paste(outcome,period,"IPCW_same_sample_unweighted",sep="_"))
      save_fit(a, paste(outcome,period,"IPCW_untruncated",sep="_"), weights$raw)
      save_fit(a, paste(outcome,period,"IPCW_truncated",sep="_"), weights$truncated)
    })
  }
  rm(d, ex, raw, prepared)
  gc(verbose=FALSE)
}
